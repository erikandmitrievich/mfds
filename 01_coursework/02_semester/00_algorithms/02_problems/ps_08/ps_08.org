mutable struct Node{T}
    data::T
    color::Symbol
    left::Union{Node{T}, Nothing}
    right::Union{Node{T}, Nothing}
    parent::Union{Node{T}, Nothing}
end

mutable struct BinaryTree{T}
    root::Union{Node{T}, Nothing}
    BinaryTree{T}() where T = new{T}(nothing)
end

isred(n) = n !== nothing && n.color == :red

function left_rotate(t::BinaryTree, n::Node)
    y = n.right
    n.right = y.left
    (y.left !== nothing) && (y.left.parent = n)
    y.parent = n.parent
    if n.parent === nothing
        t.root = y
    elseif n === n.parent.left
        n.parent.left = y
    else
        n.parent.right = y
    end
    y.left = n
    n.parent = y
end

function right_rotate(t::BinaryTree, n::Node)
    y = n.left
    n.left = y.right
    (y.right !== nothing) && (y.right.parent = n)
    y.parent = n.parent
    if n.parent === nothing
        t.root = y
    elseif n === n.parent.right
        n.parent.right = y
    else
        n.parent.left = y
    end
    y.right = n
    n.parent = y
end

function rb_insert_fixup!(t::BinaryTree, z::Node)
    while isred(z.parent)
        if z.parent === z.parent.parent.left
            y = z.parent.parent.right
            if isred(y)                        # case 1
                z.parent.color = :black
                y.color = :black
                z.parent.parent.color = :red
                z = z.parent.parent
            else
                if z === z.parent.right        # case 2
                    z = z.parent
                    left_rotate(t, z)
                end
                z.parent.color = :black        # case 3
                z.parent.parent.color = :red
                right_rotate(t, z.parent.parent)
            end
        else
            y = z.parent.parent.left
            if isred(y)                        # case 1
                z.parent.color = :black
                y.color = :black
                z.parent.parent.color = :red
                z = z.parent.parent
            else
                if z === z.parent.left         # case 2
                    z = z.parent
                    right_rotate(t, z)
                end
                z.parent.color = :black        # case 3
                z.parent.parent.color = :red
                left_rotate(t, z.parent.parent)
            end
        end
    end
    t.root.color = :black
end

function rb_insert!(t::BinaryTree{T}, key::T) where T
    z = Node{T}(key, :red, nothing, nothing, nothing)
    y, x = nothing, t.root
    while x !== nothing
        y = x
        x = z.data < x.data ? x.left : x.right
    end
    z.parent = y
    if y === nothing
        t.root = z
    elseif z.data < y.data
        y.left = z
    else
        y.right = z
    end
    rb_insert_fixup!(t, z)
    return t
end

function show_tree(n, prefix="", is_left=true)
    n === nothing && return
    if n.right !== nothing
        show_tree(n.right, prefix * (is_left ? "│   " : "    "), false)
    end
    println(prefix * (is_left ? "└── " : "┌── ") *
            "$(n.data)$(n.color == :red ? "R" : "B")")
    if n.left !== nothing
        show_tree(n.left, prefix * (is_left ? "    " : "│   "), true)
    end
end

end

* Problem 74 :red_black_tree:tree_insertion:proof:
** Task
#+begin_quote
Consider a red-black tree formed by inserting $n$ nodes with RB-INSERT. Argue that if $n > 1$, the tree has at least one red node.
#+end_quote

** Solution

1. $z$ enters red, $z \neq \text{root}$ ($n > 1$)
2. Cases 2/3 exits with $z$ red. Case 1 recolors $z.p, z.\text{uncle}, z.p.p$, never $z$. The terminal root-blackening touches only the root.
3. Therefore, $z$ is red in the final tree.

* Problem 75 :red_black_tree:tree_deletion:proof:
** Task
#+begin_quote
Argue that after RB-DELETE-FIXUP executes, the root of the tree must be black.
#+end_quote

** Solution

*Invariant.* The root is black at every loop iteration.

*Base.* Root is black on entry (precondition).

*Step.* Cases 2 and 3 touch only $x.p$, $w$, or $w$'s children — never the root. Cases 1 and 4 are the only ones that can rotate $x.p$ out of the root position; in both, the new root after $\mathrm{LEFT-ROTATE}(T, x.p)$ is $w$, and $w$ is black at that moment:

- *Case 1:* line 5 sets $w$ black before the rotation.
- *Case 4:* line 18 sets $w$ to $x.p$'s color; if $x.p$ is the root, the invariant says $x.p$ is black, so $w$ is black.

*Conclusion.* Line 44 unconditionally blackens $x$. If $x = \mathrm{root}$, this blackens the root. Otherwise, the invariant gives a black root, and line 44 doesn't touch it.

* Problem 76 :red_black_tree:tree_deletion:proof:
** Task
#+begin_quote
Argue that if in RB-DELETE both $x$ and $x.p$ are red, then property 4 is restored by the call to RB-DELETE-FIXUP(T, x).
#+end_quote

** Solution

- ~x~ is red, so the loop guard ~x != T.root and x.color == BLACK~ fails immediately and the body never runs.
- Line 44 executes unconditionally: ~x.color = BLACK~.
- This eliminates the red–red edge $(x.p, x)$; no other node is touched, so no new violation is introduced. $\blacksquare$

*RB-Tree properties:*
1. Every node is either red or black.
2. The root is black.
3. Every leaf (NIL) is black.
4. If a node is red, then both its children are black.
5. For each node, all simple paths from the node to descendant leaves contain the same number of black nodes.

    data::T
    color::Symbol
    left::Union{Node{T}, Nothing}
    right::Union{Node{T}, Nothing}
    parent::Union{Node{T}, Nothing}
end

mutable struct BinaryTree{T}
    root::Union{Node{T}, Nothing}
    BinaryTree{T}() where T = new{T}(nothing)
end

isred(n) = n !== nothing && n.color == :red

function left_rotate(t::BinaryTree, n::Node)

    n.right = y.left
    (y.left !== nothing) && (y.left.parent = n)
    y.parent = n.parent
    if n.parent === nothing
        t.root = y
    elseif n === n.parent.left
        n.parent.left = y
    else
        n.parent.right = y
    end
    y.left = n
    n.parent = y
end

function right_rotate(t::BinaryTree, n::Node)
    y = n.left
    n.left = y.right
    (y.right !== nothing) && (y.right.parent = n)
    y.parent = n.parent
    if n.parent === nothing
        t.root = y
    elseif n === n.parent.right
        n.parent.right = y
    else
        n.parent.left = y
    end
    y.right = n
    n.parent = y
end

function rb_insert_fixup!(t::BinaryTree, z::Node)
    while isred(z.parent)
        if z.parent === z.parent.parent.left
            y = z.parent.parent.right
            if isred(y)                        # case 1
                z.parent.color = :black
                y.color = :black
                z.parent.parent.color = :red
                z = z.parent.parent
            else
                if z === z.parent.right        # case 2
                    z = z.parent
                    left_rotate(t, z)
                end
                z.parent.color = :black        # case 3
                z.parent.parent.color = :red
                right_rotate(t, z.parent.parent)
            end
        else
            y = z.parent.parent.left
            if isred(y)                        # case 1
                z.parent.color = :black
                y.color = :black
                z.parent.parent.color = :red
                z = z.parent.parent
            else
                if z === z.parent.left         # case 2
                    z = z.parent
                    right_rotate(t, z)
                end
                z.parent.color = :black        # case 3
                z.parent.parent.color = :red
                left_rotate(t, z.parent.parent)
            end
        end
    end
    t.root.color = :black
end

function rb_insert!(t::BinaryTree{T}, key::T) where T
    z = Node{T}(key, :red, nothing, nothing, nothing)
    y, x = nothing, t.root
    while x !== nothing
        y = x
        x = z.data < x.data ? x.left : x.right
    end
    z.parent = y
    if y === nothing
        t.root = z
    elseif z.data < y.data
        y.left = z
    else
        y.right = z
    end
    rb_insert_fixup!(t, z)
    return t
end

function tree_minimum(n::Node)
    while n.left !== nothing
        n = n.left
    end
    return n
end

function rb_transplant!(t::BinaryTree, u::Node, v::Union{Node, Nothing})
    if u.parent === nothing
        t.root = v
    elseif u === u.parent.left
        u.parent.left = v
    else
        u.parent.right = v
    end
    if v !== nothing
        v.parent = u.parent
    end
end

function rb_delete_fixup!(t::BinaryTree, x::Union{Node, Nothing}, x_parent::Union{Node, Nothing})
    while x !== t.root && !isred(x)
        if x === x_parent.left
            w = x_parent.right
            if isred(w)
                w.color = :black
                x_parent.color = :red
                left_rotate(t, x_parent)
                w = x_parent.right
            end
            if !isred(w.left) && !isred(w.right)
                w.color = :red
                x = x_parent
                x_parent = x.parent
            else
                if !isred(w.right)
                    if w.left !== nothing; w.left.color = :black; end
                    w.color = :red
                    right_rotate(t, w)
                    w = x_parent.right
                end
                w.color = x_parent.color
                x_parent.color = :black
                if w.right !== nothing; w.right.color = :black; end
                left_rotate(t, x_parent)
                x = t.root
                x_parent = nothing
            end
        else
            w = x_parent.left
            if isred(w)
                w.color = :black
                x_parent.color = :red
                right_rotate(t, x_parent)
                w = x_parent.left
            end
            if !isred(w.right) && !isred(w.left)
                w.color = :red
                x = x_parent
                x_parent = x.parent
            else
                if !isred(w.left)
                    if w.right !== nothing; w.right.color = :black; end
                    w.color = :red
                    left_rotate(t, w)
                    w = x_parent.left
                end
                w.color = x_parent.color
                x_parent.color = :black
                if w.left !== nothing; w.left.color = :black; end
                right_rotate(t, x_parent)
                x = t.root
                x_parent = nothing
            end
        end
    end
    if x !== nothing
        x.color = :black
    end
end

function rb_delete!(t::BinaryTree, z::Node)
    y = z
    y_original_color = y.color
    x = nothing
    x_parent = nothing

    if z.left === nothing
        x = z.right
        x_parent = z.parent # x might be nothing, track parent manually
        rb_transplant!(t, z, z.right)
    elseif z.right === nothing
        x = z.left
        x_parent = z.parent
        rb_transplant!(t, z, z.left)
    else
        y = tree_minimum(z.right)
        y_original_color = y.color
        x = y.right
        if y.parent === z
            x_parent = y
        else
            x_parent = y.parent
            rb_transplant!(t, y, y.right)
            y.right = z.right
            y.right.parent = y
        end
        rb_transplant!(t, z, y)
        y.left = z.left
        y.left.parent = y
        y.color = z.color
    end

    if y_original_color == :black
        rb_delete_fixup!(t, x, x_parent)
    end
end

function find_node(n::Union{Node, Nothing}, key)
    while n !== nothing && n.data != key
        n = key < n.data ? n.left : n.right
    end
    return n
end

function show_tree(n, prefix="", is_left=true)
    n === nothing && return
    if n.right !== nothing
        show_tree(n.right, prefix * (is_left ? "│   " : "    "), false)
    end
    println(prefix * (is_left ? "└── " : "┌── ") *
            "$(n.data)$(n.color == :red ? "R" : "B")")
    if n.left !== nothing
        show_tree(n.left, prefix * (is_left ? "    " : "│   "), true)
    end
end

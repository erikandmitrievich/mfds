struct Node{T}
    data::T
    next::Union{Node{T}, Nothing}
end

mutable struct LinkedStack{T}
    top::Union{Node{T}, Nothing}
    LinkedStack{T}() where T = new(nothing)
end

Base.isempty(s::LinkedStack) = s.top === nothing

function Base.length(s::LinkedStack)
    l = 0
    n = s.top
    while n !== nothing
        l += 1
        n = n.next
    end
    l
end

function Base.push!(s::LinkedStack{T}, d::T) where T
    s.top = Node(d, s.top)
    s
end

function Base.pop!(s::LinkedStack)
    isempty(s) && throw(ArgumentError("pop! on empty LinkedStack"))
    r = s.top.data
    s.top = s.top.next
    r
end

# ------------------------------------------

struct Queue{T}
    in::LinkedStack{T}
    out::LinkedStack{T}
    Queue{T}() where T = new(LinkedStack{T}(), LinkedStack{T}())
end

Base.isempty(q::Queue) = isempty(q.in) && isempty(q.out)

Base.length(q::Queue) = length(q.in) + length(q.out)

function pour!(s_1::LinkedStack{T}, s_2::LinkedStack{T}) where T
    while (!isempty(s_1))
        d = pop!(s_1)
        push!(s_2, d)
    end
end

function enqueue!(q::Queue{T}, d::T) where T
    push!(q.in, d)
    q
end

function dequeue!(q::Queue)
    isempty(q) && throw(ArgumentError("dequeue! on empty Queue"))
    (isempty(q.out)) && pour!(q.in, q.out)
    pop!(q.out)
end

Per operation:

- =enqueue!=: $\Theta(1)$ — a single push onto =in=.
- =dequeue!=: $\Theta(1)$ if =out= is non-empty; $\Theta(n)$ if =out= is empty and =pour!= must transfer all $n$ elements.

The $\Theta(n)$ bound is tight per operation but not per sequence. Each element follows a fixed path:

1. push onto =in= (in =enqueue!=),
2. pop from =in= (in =pour!=),
3. push onto =out= (in =pour!=),
4. pop from =out= (in =dequeue!=).

Steps 2 and 3 occur at most once per element. So $e$ enqueues and any number of dequeues from empty cost at most $4e$ stack operations; $m$ operations total cost $\Theta(m)$.

mutable struct Node{T}
    data::T
    next::Union{Node{T}, Nothing}
end

mutable struct LinkedQueue{T}
    head::Union{Node{T}, Nothing}
    tail::Union{Node{T}, Nothing}
    LinkedQueue{T}() where T = new(nothing, nothing)
end

Base.isempty(q::LinkedQueue) = q.head === nothing

function Base.length(q::LinkedQueue)
    l = 0
    n = q.head
    while n !== nothing
        l += 1
        n = n.next
    end
    l
end

function enqueue!(q::LinkedQueue{T}, d::T) where T
    n = Node{T}(d, nothing)
    if q.tail === nothing
        q.head = n
    else
        q.tail.next = n
    end
    q.tail = n
    q
end

function dequeue!(q::LinkedQueue)
    isempty(q) && throw(ArgumentError("dequeue! on empty LinkedQueue"))
    r = q.head.data
    q.head = q.head.next
    (q.head === nothing) && (q.tail = nothing)
    r
end

# ------------------------------------------

mutable struct Stack{T}
    q1::LinkedQueue{T}
    q2::LinkedQueue{T}
    Stack{T}() where T = new(LinkedQueue{T}(), LinkedQueue{T}())
end

Base.isempty(s::Stack) = isempty(s.q1)

Base.length(s::Stack) = length(s.q1)

function Base.push!(s::Stack{T}, d::T) where T
    enqueue!(s.q1, d)
    s
end

function Base.pop!(s::Stack)
    isempty(s) && throw(ArgumentError("pop! on empty Stack"))
    while true
        d = dequeue!(s.q1)
        if isempty(s.q1)
            s.q1, s.q2 = s.q2, s.q1
            return d
        end
        enqueue!(s.q2, d)
    end
end

Per operation, with $n = |s|$ at the time of the call:

- =push!=: $\Theta(1)$ — one enqueue onto =q1=.
- =pop!=: $\Theta(n)$ — dequeue and re-enqueue $n-1$ elements onto =q2=, dequeue the last from =q1=, swap.

A sequence of $n$ pushes followed by $n$ pops costs $\sum_{k=1}^{n} k = \Theta(n^2)$: each =pop!= re-shuffles the survivors, and the next =pop!= re-shuffles them again — no progress is preserved between calls.

The asymmetry is structural. The mirror choice — $\Theta(n)$ =push!=, $\Theta(1)$ =pop!= — gives the same $\Theta(n^2)$ bound.

* Problem 54 :linked_list:insertion:deletion:

** Task

#+begin_quote
Explain why the dynamic-set operation INSERT on a singly linked list can be implemented in $\Theta(1)$ time, but the worst-case time for DELETE is $\Theta(n)$.
#+end_quote

mutable struct Node{T}
    data::T
    next::Union{Node{T}, Nothing}
end

mutable struct LinkedList{T}
    head::Union{Node{T}, Nothing}
    LinkedList{T}() where T = new(nothing)
end

Base.isempty(l::LinkedList) = l.head === nothing

function insert!(l::LinkedList{T}, d::T) where T
    l.head = Node{T}(d, l.head)
    l
end

function delete!(l::LinkedList{T}, n::Node{T}) where T
    if (l.head === n)
        l.head = l.head.next
    else
        p = l.head
        while (p !== nothing && p.next !== n)
            p = p.next
        end
        (p === nothing) && throw(ArgumentError("node not in list"))
        p.next = n.next
    end
    l
end

CLRS's INSERT takes a value; DELETE takes a *pointer to a node*. The asymmetry is in what is specified, not in the work itself.

- =insert!=: allocate one node, reassign =L.head=. Both steps $\Theta(1)$, independent of $n$.

- =delete!=: splicing out $x$ requires writing $x$'s predecessor's =next= field to skip $x$. A singly linked list stores no back-pointer from $x$ to its predecessor, so the predecessor must be located by walking from =L.head=. Worst case (when $x$ is the tail) this is
  $\Theta(n)$.

Even with $x$ "in hand", single linkage forces a linear lookup of its predecessor. A doubly linked list eliminates the lookup — =x.prev= gives the predecessor in $\Theta(1)$ — and DELETE drops to $\Theta(1)$.

mutable struct Node{T}
    data::T
    next::Union{Node{T}, Nothing}
end

mutable struct LinkedList{T}
    head::Union{Node{T}, Nothing}
    LinkedList{T}() where T = new(nothing)
end

Base.isempty(l::LinkedList) = l.head === nothing

function insert!(l::LinkedList{T}, d::T) where T
    l.head = Node{T}(d, l.head)
    l
end

function delete!(l::LinkedList{T}, n::Node{T}) where T
    if (l.head === n)
        l.head = l.head.next
    else
        p = l.head
        while (p !== nothing && p.next !== n)
            p = p.next
        end
        (p === nothing) && throw(ArgumentError("node not in list"))
        p.next = n.next
    end
    l
end

function Base.reverse!(l::LinkedList)
    prev = nothing
    curr = l.head

    while curr !== nothing
        next = curr.next
        curr.next = prev
        prev = curr
        curr = next
    end
    l.head = prev
    l
end

* Problem 58 :binary_tree:traversal:

** Task

#+begin_quote
Write an $O(n)$ -time nonrecursive procedure that, given an $n$ -node binary tree, prints out the key of each node. Use no more than constant extra space outside of the tree itself and do not modify the tree, even temporarily, during the procedure.
#+end_quote

** Solution

#+begin_src julia :results output
mutable struct Node{T}
    data::T
    parent::Union{Node{T}, Nothing}
    left::Union{Node{T}, Nothing}
    right::Union{Node{T}, Nothing}
end

mutable struct BinaryTree{T}
    root::Union{Node{T}, Nothing}
    BinaryTree{T}() where T = new(nothing)
end

Base.isempty(b::BinaryTree) = b.root === nothing

function wire_parents!(n::Union{Node, Nothing}, p::Union{Node, Nothing}=nothing)
    n === nothing && return
    n.parent = p
    wire_parents!(n.left, n)
    wire_parents!(n.right, n)
end

function inorder(io::IO, b::BinaryTree)
    isempty(b) && return
    prev = nothing
    curr = b.root
    while curr !== nothing
        if prev === nothing || prev === curr.parent
            # descending from parent
            if curr.left !== nothing
                next = curr.left
            else
                # no left subtree — left work is done, visit now
                print(io, curr.data, " ")
                next = curr.right !== nothing ? curr.right : curr.parent
            end
        elseif prev === curr.left
            # returned from left subtree — visit now
            print(io, curr.data, " ")
            next = curr.right !== nothing ? curr.right : curr.parent
        else  # prev === curr.right
            # returned from right subtree — go up
            next = curr.parent
        end
        prev = curr
        curr = next
    end
end

* Problem 59 :universal_hashing:

** Task

#+begin_quote
Show that any $\varepsilon$ -universal family $H$ of hash functions from a finite set $U$ to a finite set $Q$ has
$$
\varepsilon \geq \frac{1}{|Q|} - \frac{1}{|U|}
$$
#+end_quote

** Solution

Let $U$ be finite set of keys, such that $n = |U|$.
Let $Q$ be finite set of hashes, such that $m = |Q|$.
Let $H$ be a family of hash functions, such that $\forall h \in H, \, h : U \rightarrow Q$.
Let $H$ be $\epsilon$ -universal, such that $\forall k_1, k_2 \in U : k_1 \neq k_2, P(h(k_1) = h(k_2)) \le \epsilon$ for some uniformly drawn $h \in H$.

Choose uniformly $k_1, k_2$ from $U$ (allowing $k_1 = k_2$).
Choose uniformly $h$ from $H$.

*** Lower bound

For fixed $h$:

$$p_q = \dfrac{|\{k \in U : h(k) = q\}|}{|U|} \qquad (\text{probability of } k \text{ landing in } q \text{ under } h)$$

Thus,

- Each $p_q \ge 0$
- $\sum_{q\in Q} p_q = 1$

Hence,

$$P_{k_1, k_2}\Big[h(k_1) = h(k_2)\Big] = \sum_{q \in Q}P\Big[h(k_1) = q \text{ and } h(k_2) = q\Big] = \sum_{q \in Q} p_q^2 \qquad (\text{independence})$$

Using Cauchy–Schwarz:

\begin{align*}
1 = \Big(\sum_{q \in Q} p_q\Big)^2 = \Big(\sum_{q \in Q} 1 \cdot p_q\Big)^2 &\le \Big(\sum_{q \in Q} 1^2\Big) \Big(\sum_{q \in Q} p_q^2\Big) = m \cdot \sum_{q \in Q} p_q^2 \\
\dfrac{1}{m} &\le \sum_{q \in Q} p_q^2 \\
\dfrac{1}{m} &\le P_{k_1, k_2}\Big[h(k_1) = h(k_2)\Big]  \\
\end{align*}

Since this holds for every $h$, averaging over $h$ preserves it:

$$\dfrac{1}{m} \le P\Big[h(k_1) = h(k_2)\Big]$$

*** Upper bound

$$P \Big[h(k_1) = h(k_2)\Big] = \underbrace{P\Big[k_1 = k_2, h(k_1) = h(k_2)\Big]}_{(A)} + \underbrace{P\Big[k_1 \neq k_2, h(k_1) = h(k_2)\Big]}_{(B)}$$

- Term $(A)$:

  $$k_1 = k_2 \implies h(k_1) = h(k_2)$$

  Thus,

  $$(A) = P\Big[k_1 = k_2\Big] = \dfrac{1}{n}$$

- Term $(B)$:

  $$(B) = \underbrace{P\Big[k_1 \neq k_2\Big]}_{\frac{n-1}{n}\le 1} \cdot \underbrace{P\Big[h(k_1) = h(k_2) \mid k_1 \neq k_2\Big]}_{\le \epsilon} \le \epsilon$$

Combine terms:

\begin{align*}
P \Big[h(k_1) = h(k_2)\Big] &= (A) + (B) \le \dfrac{1}{n} + \epsilon
\end{align*}

*** Combine

Combining lower and upper bounds:

\begin{align*}
\dfrac{1}{m} \le P\Big[h(k_1) &= h(k_2)\Big] \le \dfrac{1}{n} + \epsilon \\
\dfrac{1}{m} &\le \dfrac{1}{n} + \epsilon \\
\dfrac{1}{m} - \dfrac{1}{n} &\le \epsilon \\
\dfrac{1}{|Q|} - \dfrac{1}{|U|} &\le \epsilon  \quad (n = |U|, m = |Q|) \\
\end{align*}

* Problem 60 :double_hashing:

** Task

#+begin_quote
Show that, with double hashing, if $m$ and $h_2(k)$ have greatest common divisor $d > 1$ for some key $k$, then an unsuccessful search for key $k$ examines $\frac{1}{d}$ th of the hash table before returning to slot $h_1(k)$. Thus, when $d = 1$, so that $m$ and $h_2(k)$ are relatively prime, the search may examine the entire hash table.
#+end_quote

** Solution

*Claim.*

Let $d = \gcd(m, h_2(k))$. The smallest positive integer $i$ such that $h(i,k) = h(0,k) \mod m$ is $i = \frac{m}{d}$. Hence, an unsuccessful search examines exactly $\frac{m}{d} = \frac{1}{d} \cdot m$ slots before returning to $h_1(k)$.

*Proof.*

The probe sequence returns to $h_1(k)$ at step $i$ iff:

$$h_1(k) + i \cdot h_2(k) \equiv h_1(k) \mod m \iff i \cdot h_2(k) \equiv 0 \mod m \qquad (*)$$

We find the smallest positive $i$ satisfying $(*)$.

Write $m = d \cdot m'$ and $h_2(k) = d \cdot h'$ with $\gcd(m', h') = 1$. Substituting into $(*)$:

$$m \mid i \cdot h_2(k) \iff d m' \mid i \cdot d h' \iff m' \mid i \cdot h'$$

Since $\gcd(m', h') = 1$, this forces $m' \mid i$. The smallest positive such $i$ is $i = m' = \frac{m}{d}$, and it satisfies $(*)$:

$$\frac{m}{d} \cdot h_2(k) = \frac{m}{d} \cdot d h' = m h' \equiv 0 \mod m$$

Therefore the probe sequence visits exactly $\frac{m}{d}$ distinct slots — one $d$ -th of the table — before returning to $h_1(k)$. $\blacksquare$

*Corollary.*

When $d = 1$, the smallest such $i$ is $m$, so the probe sequence examines every slot. This is why $m$ and $h_2(k)$ must be relatively prime — typically enforced by choosing $m$ prime and $h_2(k) < m$.

* Problem 61 :open_addressing:hashing:

** Task

#+begin_quote
Consider an open-address hash table with independent uniform permutation hashing and no deletions. Give upper bounds on the expected number of probes in an unsuccessful search and on the expected number of probes in a successful search when the load factor is $3/4$ and when it is $7/8$.
#+end_quote

** Solution

*Setup.*

Under independent uniform permutation hashing, the standard CLRS bounds apply. Let $\alpha = n/m$ be the load factor.

- Unsuccessful search: $\mathbb{E}[\text{probes}] \leq \dfrac{1}{1 - \alpha}$
- Successful search: $\mathbb{E}[\text{probes}] \leq \dfrac{1}{\alpha} \ln \dfrac{1}{1 - \alpha}$

*Case $\alpha = 3/4$.*

Unsuccessful:

$$\mathbb{E}[\text{probes}] \leq \frac{1}{1 - 3/4} = 4$$

Successful:

$$\mathbb{E}[\text{probes}] \leq \frac{4}{3} \ln 4 = \frac{8}{3} \ln 2 \approx 1.848$$

*Case $\alpha = 7/8$.*

Unsuccessful:

$$\mathbb{E}[\text{probes}] \leq \frac{1}{1 - 7/8} = 8$$

Successful:

$$\mathbb{E}[\text{probes}] \leq \frac{8}{7} \ln 8 = \frac{24}{7} \ln 2 \approx 2.377$$

*Remark.*

Doubling the empty fraction (from $1/4$ to $1/8$) doubles the unsuccessful cost ($4 \to 8$) but only adds $\approx 0.5$ to the successful cost. The unsuccessful bound is the operative constraint when sizing a table: the cliff at $\alpha \to 1$ is hyperbolic, while the successful cost grows logarithmically.

Let $f : \mathbb{R}^2 \to \mathbb{R}$ be defined by $f(x, y) = x^2 + y^2$.

(a) Compute $\nabla f(x, y)$.

(b) Evaluate it at $(1, 2)$ and determine the local (affine) linear approximation of $f$ in $(1, 2)$.

(c) Plot $f$ and $\nabla f$ in Julia.
#+end_quote

** Solution

*** (a)

\begin{align*}
\nabla f(x,y) =
\begin{pmatrix}
2x \\
2y
\end{pmatrix}
\end{align*}

*** (b)

\begin{align*}
\nabla f(1,2) =
\begin{pmatrix}
2 \\
4
\end{pmatrix}
\end{align*}

Local affine linear approximation of $f : \mathbb{R}^n \rightarrow \mathbb{R}$ in $x_0$ is

$$L_f(x) = f(x_0) + \mathrm{grad} f(x_0)(x - x_0)$$

\begin{align*}
L_f(x) &=
5 +
\begin{pmatrix}
2 & 4
\end{pmatrix}
\begin{pmatrix}
x - 2 \\
y - 4
\end{pmatrix} \\
&= 5 + 2(x - 1) + 4(y - 2) \\
&= 2x + 4y - 5
\end{align*}

function length_by_partition(f, P = range(0, 1, length=1000))
    l = 0.0
    for i in 1:(length(P) - 1)

    end

#+begin_src julia :results value
using Plots

d(x, y) = sqrt((x[1] - y[1])^2 + (x[2] - y[2])^2)

function length_by_partition(f, P = range(0, 1, length=1000))
    l = 0.0
    for i in 1:(length(P) - 1)
        l += d(f(P[i]), f(P[i+1]))
    end
    return l
end

function mid_point(f, P = range(0, 1, length=1000))
    area = 0.0

    for i in 1:(length(P) - 1)
        mid = (P[i] + P[i+1]) / 2
        h = f(mid)
        area += h * (P[i+1] - P[i])
    end
    return area
end

function length_by_midpoint(f′, P = range(0, 1, length=1000))
    s(t) = sqrt(f′(t)[1]^2 + f′(t)[2]^2)
    return mid_point(s, P)
end

# curves
f_1(t) = (cos(2π*t), sin(2π*t))
f_2(t) = (t*cos(4π*t), t*sin(4π*t))
f_3(t) = (t^2, t^3)
f_4(t) = (t*cos(6π*t), t*sin(6π*t))

# derivatives
f_1′(t) = (-2π*sin(2π*t), 2π*cos(2π*t))
f_2′(t) = (cos(4π*t) - 4π*t*sin(4π*t), sin(4π*t) + 4π*t*cos(4π*t))
f_3′(t) = (2t, 3t^2)
f_4′(t) = (cos(6π*t) - 6π*t*sin(6π*t), sin(6π*t) + 6π*t*cos(6π*t))

function main()
    curves = [(f_1, f_1′, "Unit circle"),
              (f_2, f_2′, "Spiral 1"),
              (f_3, f_3′, "Cusp curve"),
              (f_4, f_4′, "Spiral 2")]

    # print comparison
    for (f, f′, name) in curves
        a = length_by_partition(f)
        b = length_by_midpoint(f′)
        println("$name: polygonal = $a, midpoint = $b")
    end

    # plot each curve
    ts = range(0, 1, length=500)
    p = plot(layout=(2, 2), size=(900, 900), legend=false)
    for (i, (f, _, name)) in enumerate(curves)
        xs = [f(t)[1] for t in ts]
        ys = [f(t)[2] for t in ts]
        plot!(p[i], xs, ys, title=name, aspect_ratio=:equal)
    end
    savefig("./assets/ps_03_02.png")
end

main()
